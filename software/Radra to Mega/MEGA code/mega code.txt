// SETUP

void setup()
{
  // ----------------------------------------------------------
  // Laptop USB
  // ----------------------------------------------------------

  RADAR_SERIAL.begin(115200);

  // ----------------------------------------------------------
  // HC-05
  // ----------------------------------------------------------

  HC05_SERIAL.begin(HC05_BAUD);

  // ----------------------------------------------------------
  // DFPlayer
  // ----------------------------------------------------------

  DFPLAYER_SERIAL.begin(DFPLAYER_BAUD);

  // ----------------------------------------------------------
  // I2C
  // ----------------------------------------------------------

  Wire.begin();

  // ----------------------------------------------------------
  // OLED
  // ----------------------------------------------------------

  if (!display.begin(
        SSD1306_SWITCHCAPVCC,
        OLED_ADDRESS))
  {
    while (1)
    {
    }
  }

  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);

  display.setTextSize(2);
  display.setCursor(25, 5);
  display.println("VEGA");

  display.setTextSize(1);
  display.setCursor(18, 30);
  display.println("FIREFIGHTER");

  display.setCursor(25, 43);
  display.println("SAFETY SYSTEM");

  display.display();

  delay(2000);

  // ----------------------------------------------------------
  // MPU6050
  // ----------------------------------------------------------

  writeMPURegister(0x6B, 0x00);

  // Accelerometer ±8g
  writeMPURegister(0x1C, 0x10);

  // Gyroscope ±500 dps
  writeMPURegister(0x1B, 0x08);

  // ----------------------------------------------------------
  // DFPlayer initialization
  // ----------------------------------------------------------

  if (dfPlayer.begin(DFPLAYER_SERIAL))
  {
    dfPlayerAvailable = true;

    dfPlayer.volume(25);

    dfPlayer.EQ(DFPLAYER_EQ_NORMAL);

    delay(500);

    RADAR_SERIAL.println("DFPlayer: READY");
  }
  else
  {
    dfPlayerAvailable = false;

    RADAR_SERIAL.println("DFPlayer: NOT FOUND");
  }

  // ----------------------------------------------------------
  // Initial OLED
  // ----------------------------------------------------------

  showHumanPage();

  // ----------------------------------------------------------
  // Startup message
  // ----------------------------------------------------------

  RADAR_SERIAL.println();
  RADAR_SERIAL.println("=================================");
  RADAR_SERIAL.println(" VEGA MEGA 2560 READY");
  RADAR_SERIAL.println("=================================");
  RADAR_SERIAL.println("USB Radar Interface : READY");
  RADAR_SERIAL.println("OLED                : READY");
  RADAR_SERIAL.println("MLX90614            : READY");
  RADAR_SERIAL.println("MQ135               : READY");
  RADAR_SERIAL.println("MPU6050              : READY");

  if (dfPlayerAvailable)
  {
    RADAR_SERIAL.println("DFPlayer            : READY");
  }
  else
  {
    RADAR_SERIAL.println("DFPlayer            : NOT FOUND");
  }

  RADAR_SERIAL.println("=================================");
  RADAR_SERIAL.println("Waiting for radar data...");
}

// ============================================================
// LOOP
// ============================================================

void loop()
{
  // ==========================================================
  // 1. RECEIVE RADAR DATA
  // ==========================================================

  while (RADAR_SERIAL.available() > 0)
  {
    char incomingChar = RADAR_SERIAL.read();

    if (incomingChar == '\n')
    {
      processRadarMessage(radarLine);

      radarLine = "";
    }

    else if (incomingChar == '\r')
    {
      // Ignore CR
    }

    else
    {
      radarLine += incomingChar;

      if (radarLine.length() > 100)
      {
        radarLine = "";
      }
    }
  }

  // ==========================================================
  // 2. MLX90614
  // ==========================================================

  if (millis() - lastTemperatureRead >= TEMPERATURE_INTERVAL)
  {
    lastTemperatureRead = millis();

    readMLX90614();
  }

  // ==========================================================
  // 3. MQ135
  // ==========================================================

  if (millis() - lastAirRead >= AIR_INTERVAL)
  {
    lastAirRead = millis();

    readMQ135();
  }

  // ==========================================================
  // 4. MPU6050
  // ==========================================================

  if (millis() - lastMPURead >= MPU_INTERVAL)
  {
    lastMPURead = millis();

    readMPU6050();

    updateFallDetection();
  }

  // ==========================================================
  // 5. OLED PAGE ROTATION
  // ==========================================================

  if (millis() - lastDisplayChange >= DISPLAY_PAGE_INTERVAL)
  {
    lastDisplayChange = millis();

    displayPage++;

    if (displayPage > 3)
    {
      displayPage = 0;
    }

    if (displayPage == 0)
    {
      showHumanPage();
    }

    else if (displayPage == 1)
    {
      showMotionPage();
    }

    else if (displayPage == 2)
    {
      showEnvironmentPage();
    }

    else if (displayPage == 3)
    {
      showSafetyPage();
    }
  }

  // ==========================================================
  // 6. BLUETOOTH
  // ==========================================================

  if (millis() - lastBluetoothSend >= BLUETOOTH_INTERVAL)
  {
    lastBluetoothSend = millis();

    sendBluetoothData();
  }
}

// ============================================================
// PROCESS RADAR MESSAGE
// ============================================================

void processRadarMessage(String message)
{
  message.trim();

  if (message.length() == 0)
  {
    return;
  }

  // ==========================================================
  // HUMAN MESSAGE
  //
  // H,count,distance,angle,velocity
  //
  // Example:
  //
  // H,2,3.25,-10.50,0.40
  //
  // ==========================================================

  if (message.charAt(0) == 'H')
  {
    int comma1 = message.indexOf(',');
    int comma2 = message.indexOf(',', comma1 + 1);
    int comma3 = message.indexOf(',', comma2 + 1);
    int comma4 = message.indexOf(',', comma3 + 1);

    if (comma1 == -1 ||
        comma2 == -1 ||
        comma3 == -1 ||
        comma4 == -1)
    {
      return;
    }

    String countString =
        message.substring(comma1 + 1, comma2);

    String distanceString =
        message.substring(comma2 + 1, comma3);

    String angleString =
        message.substring(comma3 + 1, comma4);

    String velocityString =
        message.substring(comma4 + 1);

    // --------------------------------------------------------
    // Update radar variables
    // --------------------------------------------------------

    humanCount =
        countString.toInt();

    humanDistance =
        distanceString.toFloat();

    humanAngle =
        angleString.toFloat();

    humanVelocity =
        velocityString.toFloat();

    humanDetected =
        (humanCount > 0);

    // --------------------------------------------------------
    // Print radar information
    // --------------------------------------------------------

    RADAR_SERIAL.println();
    RADAR_SERIAL.println("---------- RADAR ----------");

    RADAR_SERIAL.print("Human Count : ");
    RADAR_SERIAL.println(humanCount);

    RADAR_SERIAL.print("Distance    : ");
    RADAR_SERIAL.print(humanDistance, 2);
    RADAR_SERIAL.println(" m");

    RADAR_SERIAL.print("Angle       : ");
    RADAR_SERIAL.print(humanAngle, 2);
    RADAR_SERIAL.println(" deg");

    RADAR_SERIAL.print("Velocity    : ");
    RADAR_SERIAL.print(humanVelocity, 2);
    RADAR_SERIAL.println(" m/s");

    RADAR_SERIAL.println("---------------------------");

    // --------------------------------------------------------
    // Update OLED immediately
    // --------------------------------------------------------

    if (displayPage == 0)
    {
      showHumanPage();
    }

    // --------------------------------------------------------
    // HUMAN ANNOUNCEMENT
    // --------------------------------------------------------
    //
    // Only announce when:
    //
    // 1. Human appears for first time
    // OR
    // 2. Human count changes
    // OR
    // 3. Distance changes significantly
    //
    // --------------------------------------------------------

    bool newHuman =
        humanDetected &&
        !previousHumanDetected;

    bool countChanged =
        (humanCount != previousHumanCount);

    bool distanceChanged =
        (fabs(humanDistance - previousHumanDistance) >= 0.5);

    bool enoughTimePassed =
        (millis() - lastAnnouncement >=
         ANNOUNCEMENT_INTERVAL);

    if (humanDetected)
    {
      if ((newHuman ||
           countChanged ||
           distanceChanged) &&
          enoughTimePassed)
      {
        playHumanAnnouncement();

        lastAnnouncement = millis();
      }
    }

    // --------------------------------------------------------
    // Save previous values
    // --------------------------------------------------------

    previousHumanDetected =
        humanDetected;

    previousHumanCount =
        humanCount;

    previousHumanDistance =
        humanDistance;
  }

  // ==========================================================
  // NO HUMAN
  //
  // N,0,0,0,0
  // ==========================================================

  else if (message.charAt(0) == 'N')
  {
    humanCount = 0;

    humanDistance = 0;
    humanAngle = 0;
    humanVelocity = 0;

    humanDetected = false;

    RADAR_SERIAL.println("NO HUMAN DETECTED");

    if (displayPage == 0)
    {
      showHumanPage();
    }

    // --------------------------------------------------------
    // Announce no human only when state changes
    // --------------------------------------------------------

    if (previousHumanDetected)
    {
      playNoHumanAlert();
    }

    previousHumanDetected = false;
    previousHumanCount = 0;
    previousHumanDistance = 0;
  }
}

void playHumanAnnouncement()
{
  if (!dfPlayerAvailable)

    return;
  }

  announcementRunning = true;

  // ----------------------------------------------------------
  // 1. HUMAN DETECTED
  // ----------------------------------------------------------

  dfPlayer.play(AUDIO_HUMAN);

  delay(1200);

  // ----------------------------------------------------------
  // 2. HUMAN COUNT
  // ----------------------------------------------------------

  playNumber(humanCount);

  delay(700);

  // ----------------------------------------------------------
  // 3. DISTANCE
  // ----------------------------------------------------------

  playDistance(humanDistance);

  delay(1000);

  announcementRunning = false;
}

// ============================================================
// PLAY NUMBER
// ============================================================
//
// 1  -> 0005
// 2  -> 0006
// 3  -> 0007
// ...
// 9  -> 0013
// 0  -> 0014
//
// ============================================================

void playNumber(int number)
{
  if (!dfPlayerAvailable)
  {
    return;
  }

  if (number < 0)
  {
    return;
  }

  // ----------------------------------------------------------
  // Zero
  // ----------------------------------------------------------

  if (number == 0)
  {
    dfPlayer.play(AUDIO_ZERO);

    delay(500);

    return;
  }

  // ----------------------------------------------------------
  // For human count, normally 1-5
  // ----------------------------------------------------------

  if (number >= 1 && number <= 9)
  {
    int audioNumber =
        AUDIO_ONE + (number - 1);

    dfPlayer.play(audioNumber);

    delay(500);

    return;
  }

  // ----------------------------------------------------------
  // For numbers greater than 9
  //
  // Example:
  //
  // 12
  //
  // One
  // Two
  //
  // ----------------------------------------------------------

  int tens = number / 10;
  int ones = number % 10;

  if (tens >= 1 && tens <= 9)
  {
    int tensAudio =
        AUDIO_ONE + (tens - 1);

    dfPlayer.play(tensAudio);

    delay(500);
  }

  if (ones > 0)
  {
    int onesAudio =
        AUDIO_ONE + (ones - 1);

    dfPlayer.play(onesAudio);

    delay(500);
  }
}

// ============================================================
// PLAY DISTANCE
// ============================================================
//
// Example:
//
// 3.25 m
//
// → Three
// → Point
// → Two
// → Five
// → Meters
//
// ============================================================

void playDistance(float distance)
{
  if (!dfPlayerAvailable)
  {
    return;
  }

  // ----------------------------------------------------------
  // Limit distance
  // ----------------------------------------------------------

  if (distance < 0)
  {
    distance = 0;
  }

  if (distance > 99.9)
  {
    distance = 99.9;
  }

  // ----------------------------------------------------------
  // Convert to one decimal place
  //
  // Example:
  //
  // 3.25 → 3.3
  //
  // ----------------------------------------------------------

  int wholePart =
      (int)distance;

  int decimalPart =
      (int)((distance - wholePart) * 10.0 + 0.5);

  // ----------------------------------------------------------
  // Handle rounding
  // ----------------------------------------------------------

  if (decimalPart >= 10)
  {
    wholePart++;
    decimalPart = 0;
  }

  // ----------------------------------------------------------
  // Whole number
  // ----------------------------------------------------------

  playNumber(wholePart);

  delay(400);

  // ----------------------------------------------------------
  // Decimal point
  // ----------------------------------------------------------

  if (decimalPart > 0)
  {
    dfPlayer.play(AUDIO_POINT);

    delay(400);

    playNumber(decimalPart);

    delay(400);
  }

  // ----------------------------------------------------------
  // Meters
  // ----------------------------------------------------------

  dfPlayer.play(AUDIO_METERS);

  delay(600);
}

// ============================================================
// MLX90614 READ
// ============================================================

float readMLXTemperature(uint8_t registerAddress)
{
  Wire.beginTransmission(MLX90614_ADDRESS);

  Wire.write(registerAddress);

  if (Wire.endTransmission(false) != 0)
  {
    return NAN;
  }

  Wire.requestFrom(
    MLX90614_ADDRESS,
    (uint8_t)3
  );

  if (Wire.available() < 3)
  {
    return NAN;
  }

  uint8_t lowByte = Wire.read();
  uint8_t highByte = Wire.read();

  // PEC byte
  Wire.read();

  uint16_t rawValue =
      ((uint16_t)highByte << 8) |
      lowByte;

  float temperature =
      (rawValue * 0.02) - 273.15;

  return temperature;
}

// ============================================================
// READ MLX90614
// ============================================================

void readMLX90614()
{
  float ambient =
      readMLXTemperature(0x06);

  float object =
      readMLXTemperature(0x07);

  if (!isnan(ambient))
  {
    ambientTemperature = ambient;
  }

  if (!isnan(object))
  {
    objectTemperature = object;
  }
}

// ============================================================
// READ MQ135
// ============================================================

void readMQ135()
{
  int adcValue =
      analogRead(MQ135_PIN);

  if (adcValue <= 0)
  {
    airPPM = 0;
    return;
  }

  float voltage =
      adcValue * (5.0 / 1023.0);

  if (voltage <= 0.01)
  {
    airPPM = 0;
    return;
  }

  float sensorResistance =
      ((5.0 - voltage) / voltage) * 10.0;

  if (sensorResistance <= 0)
  {
    airPPM = 0;
    return;
  }

  // Approximate model.
  // NOT laboratory calibrated.

  airPPM =
      1000.0 *
      (10.0 / sensorResistance);
}

{

}

void readMPUData()
{
  Wire.beginTransmission(
    MPU6050_ADDRESS
  );

  Wire.write(0x3B);

  Wire.endTransmission(false);

  Wire.requestFrom(
    MPU6050_ADDRESS,
    (uint8_t)14
  );

  if (Wire.available() < 14)
  {
    return;
  }

  accelX =
      ((int16_t)Wire.read() << 8) |
      Wire.read();

  accelY =
      ((int16_t)Wire.read() << 8) |
      Wire.read();

  accelZ =
      ((int16_t)Wire.read() << 8) |
      Wire.read();

  // Temperature
  Wire.read();
  Wire.read();

  gyroX =
      ((int16_t)Wire.read() << 8) |
      Wire.read();

  gyroY =
      ((int16_t)Wire.read() << 8) |
      Wire.read();

  gyroZ =
      ((int16_t)Wire.read() << 8) |
      Wire.read();
}

// ============================================================
// READ MPU6050
// ============================================================

void readMPU6050()
{
  readMPUData();

  // ±8g
  ax_g =
      accelX / 4096.0;

  ay_g =
      accelY / 4096.0;

  az_g =
      accelZ / 4096.0;

  // ±500 dps
  gx_dps =
      gyroX / 65.5;

  gy_dps =
      gyroY / 65.5;

  gz_dps =
      gyroZ / 65.5;

  totalAcceleration =
      sqrt(
        ax_g * ax_g +
        ay_g * ay_g +
        az_g * az_g
      );

  float normalizedZ =
      az_g /
      max(totalAcceleration, 0.001f);

  normalizedZ =
      constrain(
        normalizedZ,
        -1.0,
        1.0
      );

  tiltAngle =
      acos(normalizedZ)
      * 180.0
      / PI;
}

// ============================================================
// FALL DETECTION
// ============================================================

void updateFallDetection()
{
  unsigned long now =
      millis();

  // ==========================================================
  // ALREADY DETECTED
  // ==========================================================

  if (fallState == FALL_DETECTED)
  {
    fallAlert = true;

    if (tiltAngle < RECOVERY_TILT)
    {
      if (recoveryStart == 0)
      {
        recoveryStart = now;
      }

      if (now - recoveryStart >= RECOVERY_TIME)
      {
        fallState = FALL_NORMAL;

        fallAlert = false;

        recoveryStart = 0;
      }
    }
    else
    {
      recoveryStart = 0;
    }

    return;
  }

  // ==========================================================
  // FREE FALL
  // ==========================================================

  if (totalAcceleration <
      FREE_FALL_THRESHOLD)
  {
    if (freeFallStart == 0)
    {
      freeFallStart = now;
    }

    fallState =
        FALL_FREE_FALL;
  }

  // ==========================================================
  // IMPACT
  // ==========================================================

  else if (fallState ==
           FALL_FREE_FALL)
  {
    if (totalAcceleration >=
        IMPACT_THRESHOLD)
    {
      impactStart = now;

      fallState =
          FALL_IMPACT;
    }

    else if (now - freeFallStart > 1500)
    {
      fallState =
          FALL_NORMAL;

      freeFallStart = 0;
    }
  }

  // ==========================================================
  // AFTER IMPACT → CHECK TILT
  // ==========================================================

  else if (fallState ==
           FALL_IMPACT)
  {
    if (tiltAngle >=
        TILT_THRESHOLD)
    {
      if (abnormalTiltStart == 0)
      {
        abnormalTiltStart = now;
      }

      fallState =
          FALL_TILT;

      if (now - abnormalTiltStart >=
          TILT_CONFIRM_TIME)
      {
        fallState =
            FALL_DETECTED;

        fallAlert = true;

        playFallAlert();
      }
    }
    else
    {
      fallState =
          FALL_NORMAL;

      freeFallStart = 0;
      abnormalTiltStart = 0;
    }
  }

  // ==========================================================
  // ABNORMAL TILT
  // ==========================================================

  else if (fallState ==
           FALL_TILT)
  {
    if (tiltAngle >=
        TILT_THRESHOLD)
    {
      if (now - abnormalTiltStart >=
          TILT_CONFIRM_TIME)
      {
        fallState =
            FALL_DETECTED;

        fallAlert = true;

        playFallAlert();
      }
    }
    else
    {
      fallState =
          FALL_NORMAL;

      abnormalTiltStart = 0;
    }
  }

  // ==========================================================
  // NORMAL
  // ==========================================================

  else
  {
    if (tiltAngle <
        TILT_THRESHOLD)
    {
      fallState =
          FALL_NORMAL;

      fallAlert = false;

      freeFallStart = 0;
      abnormalTiltStart = 0;
    }
  }
}

void sendBluetoothData()
{
  HC05_SERIAL.print("HUMAN=");
  HC05_SERIAL.print(humanCount);

  HC05_SERIAL.print(",DIST=");
  HC05_SERIAL.print(humanDistance, 2);

  HC05_SERIAL.print(",ANGLE=");
  HC05_SERIAL.print(humanAngle, 1);

  HC05_SERIAL.print(",VEL=");
  HC05_SERIAL.print(humanVelocity, 2);

  HC05_SERIAL.print(",TEMP=");
  HC05_SERIAL.print(objectTemperature, 1);

  HC05_SERIAL.print(",AIR=");
  HC05_SERIAL.print(airPPM, 0);

  HC05_SERIAL.print(",FALL=");

  if (fallAlert)
  {
    HC05_SERIAL.println("DETECTED");
  }
  else
  {
    HC05_SERIAL.println("NORMAL");
  }
}
